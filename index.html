const a=e.parameter.action; if(a==='getParts'){const sh=ss.getSheetByName(SS_PARTS)||ss.insertSheet(SS_PARTS);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getHistory'){const sh=ss.getSheetByName(SS_HIST)||ss.insertSheet(SS_HIST);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getImages'){const sh=ss.getSheetByName('Images')||ss.insertSheet('Images');return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getCheckItems'){const sh=ss.getSheetByName(SS_CHECK)||ss.insertSheet(SS_CHECK);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getCheckData'){const sh=ss.getSheetByName(SS_CHECKDATA)||ss.insertSheet(SS_CHECKDATA);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getNames'){const sh=ss.getSheetByName(SS_NAMES)||ss.insertSheet(SS_NAMES);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getToolImages'){const sh=ss.getSheetByName(SS_TOOLIMG)||ss.insertSheet(SS_TOOLIMG);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getWeeklySchedule'){const sh=ss.getSheetByName(SS_WEEKLY)||ss.insertSheet(SS_WEEKLY);return out({ok:true,data:sh.getDataRange().getValues()});} if(a==='getMonthlyStockCheck'){const sh=ss.getSheetByName(SS_MSC)||ss.insertSheet(SS_MSC);return out({ok:true,data:sh.getDataRange().getValues()});} return out({ok:false}); } function doPost(e){ const p=JSON.parse(e.postData.contents),ss=SpreadsheetApp.getActiveSpreadsheet(); if(p.action==='saveParts'){ let sh=ss.getSheetByName(SS_PARTS)||ss.insertSheet(SS_PARTS); sh.clearContents(); const h=['id','code','name','category','spec','location','qty','min','max','unit','ordering','quoting','uncontrol']; sh.getRange(1,1,1,h.length).setValues([h]); if(p.parts&&p.parts.length)sh.getRange(2,1,p.parts.length,h.length).setValues(p.parts.map(x=>[x.id,x.code,x.name,x.category,x.spec||'',x.location||'',x.qty,x.min,x.max!=null?x.max:'',x.unit,x.ordering?1:'',x.quoting?1:'',x.uncontrol?1:''])); return out({ok:true}); } if(p.action==='saveHistory'){ let sh=ss.getSheetByName(SS_HIST)||ss.insertSheet(SS_HIST); sh.clearContents(); const h=['id','partId','partName','partCode','type','qty','unit','technician','prodline','machine','note','time']; sh.getRange(1,1,1,h.length).setValues([h]); if(p.history&&p.history.length)sh.getRange(2,1,p.history.length,h.length).setValues(p.history.map(x=>[x.id,x.partId,x.partName,x.partCode||'',x.type,x.qty,x.unit,x.technician||'',x.prodline||'',x.machine||'',x.note||'',x.time])); return out({ok:true}); } if(p.action==='saveImages'){ let sh=ss.getSheetByName('Images')||ss.insertSheet('Images'); sh.clearContents(); const h=['partId','b64']; sh.getRange(1,1,1,h.length).setValues([h]); if(p.images&&p.images.length)sh.getRange(2,1,p.images.length,h.length).setValues(p.images.map(x=>[x.partId,x.b64])); return out({ok:true}); } if(p.action==='saveCheckItems'){ let sh=ss.getSheetByName(SS_CHECK)||ss.insertSheet(SS_CHECK); sh.clearContents(); sh.getRange(1,1,1,1).setValues([['json']]); sh.getRange(2,1,1,1).setValues([[JSON.stringify(p.checkItems)]]); return out({ok:true}); } if(p.action==='saveCheckData'){ saveKeyedJson(SS_CHECKDATA,p.checkData); return out({ok:true}); } if(p.action==='saveNames'){ let sh=ss.getSheetByName(SS_NAMES)||ss.insertSheet(SS_NAMES); sh.clearContents(); sh.getRange(1,1,1,1).setValues([['json']]); sh.getRange(2,1,1,1).setValues([[JSON.stringify(p.names)]]); return out({ok:true}); } if(p.action==='saveToolImages'){ let sh=ss.getSheetByName(SS_TOOLIMG)||ss.insertSheet(SS_TOOLIMG); sh.clearContents(); const h=['toolId','b64']; sh.getRange(1,1,1,h.length).setValues([h]); if(p.images&&p.images.length)sh.getRange(2,1,p.images.length,h.length).setValues(p.images.map(x=>[x.toolId,x.b64])); return out({ok:true}); } if(p.action==='saveWeeklySchedule'){ saveKeyedJson(SS_WEEKLY,p.weeklySchedule); return out({ok:true}); } if(p.action==='saveMonthlyStockCheck'){ saveKeyedJson(SS_MSC,p.monthlyStockCheck); return out({ok:true}); } return out({ok:false}); } // เก็บข้อมูลแบบ "1 แถวต่อ 1 key" (key, data_json) แทนการยัดทุกอย่างเป็น JSON ก้อนเดียวในเซลล์เดียว // เดิมยัดทุกอย่างในเซลล์เดียวแล้ว merge สะสมไปเรื่อยๆไม่มีวันลบของเก่า พอข้อมูลเยอะขึ้นความยาว JSON // จะเกินขีดจำกัด 50,000 ตัวอักษรต่อเซลล์ของ Google Sheets ทำให้ setValues() พัง (throw exception) // กลางคันหลังจากเขียน header ไปแล้ว — sheet จะโชว์ header เฉยๆ แต่แถวข้อมูลว่างเปล่าตลอดไป // วิธีนี้ไม่มีทางชนขีดจำกัดอีกเลย เพราะแต่ละเซลล์เก็บแค่ 1 record ไม่ใช่ทั้งก้อน function saveKeyedJson(sheetName,dataObj){ const ss=SpreadsheetApp.getActiveSpreadsheet(); const sh=ss.getSheetByName(sheetName)||ss.insertSheet(sheetName); const rows=sh.getDataRange().getValues(); const existing={}; for(let i=1;i{ existing[k]=JSON.stringify(dataObj[k]); }); sh.clearContents(); sh.getRange(1,1,1,2).setValues([['key','data_json']]); const keys=Object.keys(existing); if(keys.length) sh.getRange(2,1,keys.length,2).setValues(keys.map(k=>[k,existing[k]])); } function out(obj){return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);}
